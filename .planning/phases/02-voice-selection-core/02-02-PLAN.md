**Change 3: Add async voice loading JS in the `<script>` block**

Add the following JavaScript BEFORE the closing `</script>` tag (before line 830). This implements the async dual-fetch pattern per D-13 and D-10:

```javascript
// Voice dropdown async loading
var _voicesLoaded = false;\
var _lastVoiceLoaded = false;\
var _voiceList = [];\
var _lastVoiceId = '';\
\
google.script.run\
  .withSuccessHandler(function(voices) {\
    _voiceList = voices || [];\
    _voicesLoaded = true;\
    if (_lastVoiceLoaded) _populateVoiceDropdown();\
  })\
  .withFailureHandler(function(err) {\
    var sel = document.getElementById('voice_id');\
    sel.innerHTML = '<option value="">Error loading voices</option>';\
  })\
  .getVoiceList();\
\
google.script.run\
  .withSuccessHandler(function(id) {\
    _lastVoiceId = id || '';\
    _lastVoiceLoaded = true;\
    if (_voicesLoaded) _populateVoiceDropdown();\
  })\
  .withFailureHandler(function() {\
    _lastVoiceLoaded = true;\
    if (_voicesLoaded) _populateVoiceDropdown();\
  })\
  .getLastUsedVoiceId();\
\
function _populateVoiceDropdown() {\
  var sel = document.getElementById('voice_id');\
  sel.innerHTML = '<option value="">Agent default voice</option>';\
  _voiceList.forEach(function(v) {\
    var opt = document.createElement('option');\
    opt.value = v.voice_id;\
    opt.textContent = v.name + (v.description ? ' \\u2014 ' + v.description : '');\
    if (v.voice_id === _lastVoiceId) opt.selected = true;\
    sel.appendChild(opt);\
  });\
}\
```

NOTE on the em-dash: Use `\\u2014` (unicode escape) for the em-dash separator between voice name and labels (e.g., "Alice -- British, Female, Clear") per D-03. This avoids encoding issues in the inline HTML string. The `\\u2014` renders as an em-dash in the browser.

NOTE on the dual-fetch pattern: Both `getVoiceList()` and `getLastUsedVoiceId()` are called in parallel. `_populateVoiceDropdown()` only runs when BOTH have completed. This ensures last-used voice is correctly pre-selected even if it loads before the voice list.
