    ```javascript
    /**
     * Fetch one page of the authenticated LinkedIn DM inbox via Voyager.
     * Runs fetch() inside the page context so it inherits cookies/CSRF.
     * Returns { ok: true, data } on 2xx, { ok: false, status|error } otherwise.
     *
     * Endpoint: GET /voyager/api/messaging/conversations?keyVersion=LEGACY_INBOX
     * Pagination: REST-li start + count (community standard; nsandman/linkedin-api, tomquirk/linkedin-api).
     */
    export async function getConversationsPage(page, { start = 0, count = 20 } = {}) {
      return page.evaluate(async ({ start, count }) => {
        try {
          const csrf = document.cookie.split(';')
            .map(c => c.trim())
            .find(c => c.startsWith('JSESSIONID='));
          if (!csrf) return { ok: false, error: 'no_jsessionid' };
          const token = csrf.split('=')[1]?.replace(/"/g, '');

          const url = `https://www.linkedin.com/voyager/api/messaging/conversations`
            + `?keyVersion=LEGACY_INBOX&count=${count}&start=${start}`;

          const resp = await fetch(url, {
            headers: {
              'accept': 'application/vnd.linkedin.normalized+json+2.1',
              'csrf-token': token,
              'x-restli-protocol-version': '2.0.0',
              'x-li-lang': 'en_US',
            },
            credentials: 'include',
          });

          if (!resp.ok) return { ok: false, status: resp.status, url };
          const data = await resp.json();
          return { ok: true, data };
        } catch (e) {
          return { ok: false, error: String(e?.message || e) };
        }
      }, { start, count });
    }
    ```

    ```javascript
    export function candidatesForProfile(rows, profileName, linkedinColumn) {
      const wantName = (profileName || '').toString().toLowerCase().trim();
      return (rows || []).filter(row => {
        // LinkedIn URL presence (only enforce if a column name was given)
        if (linkedinColumn) {
          const url = extractLinkedInUrl(row, linkedinColumn);
          if (!url) return false;
        }
        const msgCell    = (row['Message']      || row['message']      || '').toString().toLowerCase().trim();
        const acctCell   = (row['Account Used'] || row['account used'] || '').toString().toLowerCase().trim();
        if (msgCell !== 'sent') return false;
        if (acctCell !== wantName) return false;
        return true;
      });
    }

    export function extractConversationsArray(voyagerResponse) {
      if (!voyagerResponse || typeof voyagerResponse !== 'object') return [];
      const elements = Array.isArray(voyagerResponse.elements) ? voyagerResponse.elements : [];
      const included = Array.isArray(voyagerResponse.included) ? voyagerResponse.included : [];

      // Build profile lookup from MiniProfile entities keyed by entityUrn
      const miniProfileByUrn = new Map();
      for (const ent of included) {
        const t = ent?.$type || '';
        if (t.includes('MiniProfile') && ent.entityUrn) {
          miniProfileByUrn.set(ent.entityUrn, ent);
        }
      }

      // Elements may live in `elements` or `included` depending on normalization mode
      const convEntities = elements.length ? elements : included.filter(e => (e?.$type || '').includes('Conversation'));

      const out = [];
      for (const c of convEntities) {
        try {
          const entityUrn = c.entityUrn || '';
          const threadId = entityUrn.replace(/^urn:li:fs_conversation:/, '').replace(/^urn:li:fsd_conversation:/, '');

          // Participant name — try the participants field, fall back to any linked MiniProfile
          let firstName = '', lastName = '', publicIdentifier = null;
          const participants = Array.isArray(c.participants) ? c.participants : [];
          for (const p of participants) {
            const ref = typeof p === 'string' ? p : (p?.['com.linkedin.voyager.messaging.MessagingMember']?.miniProfile || p?.miniProfile || p);
            const mini = (typeof ref === 'string') ? miniProfileByUrn.get(ref) : ref;
            if (mini?.firstName || mini?.lastName) {
              firstName = mini.firstName || '';
              lastName = mini.lastName || '';
              publicIdentifier = mini.publicIdentifier || null;
              break;
            }
          }

          // Last-activity timestamp (ms since epoch)
          const lastActivityAtMs = Number.isFinite(c.lastActivityAt) ? c.lastActivityAt
            : Number.isFinite(c.lastModifiedAt) ? c.lastModifiedAt
            : (Array.isArray(c.events) && c.events[0]?.createdAt) || 0;

          // Snippet — body text from first event, truncated
          const firstEvent = Array.isArray(c.events) ? c.events[0] : null;
          const rawText = firstEvent?.eventContent?.['com.linkedin.voyager.messaging.event.MessageEvent']?.attributedBody?.text
            || firstEvent?.body?.text
            || firstEvent?.body
            || '';
          const snippet = typeof rawText === 'string' ? rawText.slice(0, 100) : '';

          out.push({ firstName, lastName, publicIdentifier, snippet, lastActivityAtMs: Number(lastActivityAtMs) || 0, entityUrn, threadId });
        } catch {
          // Skip malformed entries — degraded fixture drives us through this branch
        }
      }
      return out;

    export function matchConversation(conv, candidates) {
      const pFull = `${normalizeName(conv?.firstName)} ${normalizeName(conv?.lastName)}`.trim();
      if (!pFull) return { matched: false, reason: 'no_participant_name' };

      const hits = (candidates || []).filter(row => {
        const rFull = `${normalizeName(row.firstName || row['firstName'])} ${normalizeName(row.lastName || row['lastName'])}`;
        return rFull === pFull;
      });

      if (hits.length === 0) return { matched: false, reason: 'no_candidate_row' };
      if (hits.length === 1) return { matched: true, row: hits[0] };

      if (conv.publicIdentifier) {
        const slug = conv.publicIdentifier.toLowerCase();
        const urlHit = hits.find(r => (r['LinkedIn URL'] || r.linkedinUrl || '').toLowerCase().includes(`/in/${slug}`));
        if (urlHit) return { matched: true, row: urlHit };
      }
      return { matched: false, reason: 'ambiguous', candidates: hits.length };
    }
    ```
