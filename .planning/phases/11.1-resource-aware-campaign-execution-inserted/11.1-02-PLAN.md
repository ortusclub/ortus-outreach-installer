test('single-profile + high RAM → loop picks 4x multiplier', () => {
  const cfg = loadConfig();
  const state = { active: false, reason: '', multiplier: 1 };
  const highRamSample = { ramPct: 92, load1: 1.0, cpuCount: 8 };
