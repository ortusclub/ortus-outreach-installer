    ```javascript
    /**
     * Fetch one page of the authenticated LinkedIn DM inbox via Voyager.
     * Runs fetch() inside the page context so it inherits cookies/CSRF.
     * Returns { ok: true, data } on 2xx, { ok: false, status|error } otherwise.
     *
     * Endpoint: GET /voyager/api/messaging/conversations?keyVersion=LEGACY_INBOX
     * Pagination: REST-li start + count (community standard; nsandman/linkedin-api, tomquirk/linkedin-api).
     */
    export async function getConversationsPage(page, { start = 0, count = 20 } = {}) {
      return page.evaluate(async ({ start, count }) => {
        try {
          const csrf = document.cookie.split(';')
            .map(c => c.trim())
            .find(c => c.startsWith('JSESSIONID='));
          if (!csrf) return { ok: false, error: 'no_jsessionid' };
          const token = csrf.split('=')[1]?.replace(/"/g, '');

          const url = `https://www.linkedin.com/voyager/api/messaging/conversations`
            + `?keyVersion=LEGACY_INBOX&count=${count}&start=${start}`;

          const resp = await fetch(url, {
            headers: {
              'accept': 'application/vnd.linkedin.normalized+json+2.1',
              'csrf-token': token,
              'x-restli-protocol-version': '2.0.0',
              'x-li-lang': 'en_US',
            },
            credentials: 'include',
          });

          if (!resp.ok) return { ok: false, status: resp.status, url };
          const data = await resp.json();
          return { ok: true, data };
        } catch (e) {
          return { ok: false, error: String(e?.message || e) };
        }
      }, { start, count });
    }
    ```

    export async function scrapeInboxDom(page) {
      try {
        await sleep(1500); // polite pacing before additional navigation
        await page.goto('https://www.linkedin.com/messaging/', {
          waitUntil: 'domcontentloaded',
          timeout: 30000,
        });
        // Wait for either the modern class-based list OR the data-test attribute variant — whichever resolves first
        await Promise.race([
          page.waitForSelector('ul.msg-conversations-container__conversations-list', { timeout: 10000 }),
          page.waitForSelector('[data-test="conversations-list"]',                  { timeout: 10000 }),
        ]);
        const html = await page.evaluate(() => {
          const el = document.querySelector('ul.msg-conversations-container__conversations-list')
                  || document.querySelector('[data-test="conversations-list"]');
          return el ? el.outerHTML : '';
        });
        if (!html) return { ok: false, error: 'inbox_list_empty' };
        return { ok: true, html };
      } catch (e) {
        return { ok: false, error: String(e?.message || e) };
      }
    }

    /**
     * DMS-03 DOM-fallback parser. Pure — takes an HTML string (from scrapeInboxDom in helpers.js),
     * returns a Conversation[] with the same shape as extractConversationsArray. Unit-testable
     * against tests/fixtures/messaging-inbox-dom.html without Puppeteer.
     *
     * Implementation uses a lightweight regex-based pass over each <li class="msg-conversation-listitem">.
     * We intentionally avoid adding jsdom/cheerio as a devDep (CLAUDE.md: no new deps without strong
     * justification). The LinkedIn inbox list item shape is small and stable enough that regex
     * extraction is acceptable. If LinkedIn rotates the markup, this file is the single place to fix
     * — a full parser swap (e.g. adopting node-html-parser if it lands in Node std) happens here, not
     * scattered across the orchestrator.
     */
    export function parseInboxDom(htmlString) {
      if (!htmlString || typeof htmlString !== 'string') return [];
      const out = [];
      // Split on list-item boundaries. Use a non-greedy match so each <li>...</li> is one conversation.
      const itemRe = /<li\s[^>]*class=["'][^"']*msg-conversation-listitem[^"']*["'][^>]*>([\s\S]*?)<\/li>/g;
      let m;
      while ((m = itemRe.exec(htmlString)) !== null) {
        try {
          const inner = m[0]; // includes the <li ...> opening tag, needed for data-urn extraction
          // data-urn="urn:li:fsd_conversation:2-ABC..."  OR  href="/messaging/thread/2-ABC.../"
          const urnMatch = inner.match(/data-urn=["']urn:li:fs[d]?_conversation:([^"']+)["']/);
          const hrefMatch = inner.match(/href=["']\/messaging\/thread\/([^"'\/]+)\/?["']/);
          const threadId = (urnMatch && urnMatch[1]) || (hrefMatch && hrefMatch[1]) || '';

          // Participant name
          const nameMatch = inner.match(/msg-conversation-listitem__participant-names[^>]*>\s*<span[^>]*>([^<]+)<\/span>/)
                         || inner.match(/msg-conversation-listitem__participant-names[^>]*>([^<]+)</);
          const fullName = nameMatch ? nameMatch[1].trim() : '';
          if (!fullName) continue; // malformed row — skip without throwing (matches the fixture's 2-MALFORMED case)
          const parts = fullName.split(/\s+/);
          const firstName = parts.slice(0, 1).join(' ');
          const lastName  = parts.slice(1).join(' ');

          // Snippet
          const snippetMatch = inner.match(/msg-conversation-listitem__message-snippet[^>]*>([\s\S]*?)<\/(?:p|span|div)>/);
          const snippet = snippetMatch ? snippetMatch[1].replace(/<[^>]+>/g, '').trim() : '';

          // Timestamp — prefer the datetime attribute on <time>
          const timeMatch = inner.match(/<time[^>]*datetime=["']([^"']+)["']/);
          const lastActivityAtMs = timeMatch ? (Date.parse(timeMatch[1]) || 0) : 0;

          out.push({
            firstName,
            lastName,
            publicIdentifier: null, // DOM path does not reliably expose this — matching degrades to name-only
            snippet,
            lastActivityAtMs,
            entityUrn: urnMatch ? `urn:li:fsd_conversation:${threadId}` : '',
            threadId,
          });
        } catch {
          // Swallow any individual row parse failure — resilience beats coverage here
        }
      }
      return out;
    }

    export function matchConversation(conv, candidates) {
      const pFull = `${normalizeName(conv?.firstName)} ${normalizeName(conv?.lastName)}`.trim();
      if (!pFull) return { matched: false, reason: 'no_participant_name' };

      const hits = (candidates || []).filter(row => {
        const rFull = `${normalizeName(row.firstName || row['firstName'])} ${normalizeName(row.lastName || row['lastName'])}`;
        return rFull === pFull;
      });

      if (hits.length === 0) return { matched: false, reason: 'no_candidate_row' };
      if (hits.length === 1) return { matched: true, row: hits[0] };

      if (conv.publicIdentifier) {
        const slug = conv.publicIdentifier.toLowerCase();
        const urlHit = hits.find(r => (r['LinkedIn URL'] || r.linkedinUrl || '').toLowerCase().includes(`/in/${slug}`));
        if (urlHit) return { matched: true, row: urlHit };
      }
      return { matched: false, reason: 'ambiguous', candidates: hits.length };
    }

    export function filterByWatermark(convs, watermarkIso) {
      if (!watermarkIso) return convs.slice();

      if (!Number.isFinite(wmMs)) return convs.slice();
      return convs.filter(c => Number.isFinite(c.lastActivityAtMs) && c.lastActivityAtMs > wmMs);
    }
