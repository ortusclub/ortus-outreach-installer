The test stands up an ephemeral Node `http.createServer`, records every POST body it receives, and drives the Check DMs write-back flow through `getSheetRowStatus` + `shouldWriteReply` + (would-be) `updateSheetRow`. It covers four critical cases:

    1. Happy path: getStatus returns Reply='' → updateSheetRow IS called
    2. Canonical skip: getStatus returns Reply='yes' → updateSheetRow is NOT called
    3. Case variants on VALUE: Reply='YES', Reply='Yes' → also NOT called
    4. Case variants on KEY (Plan 11-02 revision): returned row has `reply: 'yes'` (lowercase key) or `REPLY: 'yes'` (uppercase key) → STILL NOT called (case-insensitive header normalization from the revision)

    ```javascript
    // tests/check-dms-writeback-integration.test.js
    // Wire-level integration for DMS-04 non-destructive skip (Plan 11-02 Task 4 revision).
    //
    // This complements tests/check-dms-writeback.test.js (pure shouldWriteReply predicate)
    // by proving the END-TO-END flow:
    //   getSheetRowStatus(sheetUrl, linkedinUrl) → Apps Script getStatus call
    //     → shouldWriteReply({ existingReply }) with normalized header lookup
    //     → updateSheetRow only when shouldWriteReply returns true
    //
    // We stand up a real HTTP server that impersonates the Apps Script webapp, so the
    // normalization of row['Reply']/row.reply/row.REPLY is exercised against actual
    // JSON payloads on the wire — not just against a fixture in memory.
    import { describe, it, before, after, beforeEach } from 'node:test';
    import assert from 'node:assert/strict';
    import http from 'node:http';

    // Import the helpers under test. These are imported AFTER the server is set up
    // and SHEETS_WEBAPP_URL is configured (module-level env-var reads must see the URL).
    let getSheetRowStatus, updateSheetRow, shouldWriteReply;

    // Mock server state — reset between tests
    let server, serverUrl;
    let recordedPosts = [];
    let nextStatusResponse = null;   // what getStatus returns for the next call
    let nextUpdateResponse = null;   // what updateRow returns for the next call

    before(async () => {
      server = http.createServer((req, res) => {
        let body = '';
        req.on('data', chunk => { body += chunk; });
        req.on('end', () => {
          let parsed = {};
          try { parsed = JSON.parse(body); } catch {}
          recordedPosts.push({ action: parsed.action, body: parsed });

          res.setHeader('Content-Type', 'application/json');
          if (parsed.action === 'getStatus') {
            res.end(JSON.stringify(nextStatusResponse || { rows: [] }));
          } else if (parsed.action === 'updateRow') {
            res.end(JSON.stringify(nextUpdateResponse || { success: true }));
          } else {
            res.end(JSON.stringify({ error: 'unknown action' }));
          }
        });
      });
      await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
      const { port } = server.address();
      serverUrl = `http://127.0.0.1:${port}/`;
      process.env.SHEETS_WEBAPP_URL = serverUrl;

      // Import AFTER env-var set, so module-level getWebAppUrl() sees the mock URL.
      ({ getSheetRowStatus, updateSheetRow } = await import('../src/sheets-writer.js'));
      ({ shouldWriteReply } = await import('../src/linkedin/check-dms.js'));
    });

    after(async () => {
      await new Promise(resolve => server.close(resolve));
      delete process.env.SHEETS_WEBAPP_URL;
    });

    beforeEach(() => {
      recordedPosts = [];
      nextStatusResponse = null;
      nextUpdateResponse = null;
    });

    // Simulates the exact write-back branch from Plan 11-02 Task 4 — the case-insensitive
    // header lookup is inlined here so the test proves the normalization rule
    // end-to-end, matching the orchestrator's shape.
    async function writeBackFlow(sheetUrl, linkedinUrl, replyFields) {
      const statusRow = await getSheetRowStatus(sheetUrl, linkedinUrl);
      const existingReply = statusRow
        ? (statusRow['Reply']
           ?? statusRow.reply
           ?? (Object.keys(statusRow).find(k => k.toLowerCase() === 'reply')
               ? statusRow[Object.keys(statusRow).find(k => k.toLowerCase() === 'reply')]
               : ''))
        : '';
      if (!shouldWriteReply({ existingReply })) return { skipped: true };
      await updateSheetRow(sheetUrl, linkedinUrl, replyFields, 'LinkedIn URL');
      return { skipped: false };
    }

    const SHEET_URL = 'https://docs.google.com/spreadsheets/d/TESTID/edit';
    const PROSPECT_URL = 'https://www.linkedin.com/in/jane-doe';
    const REPLY_FIELDS = { reply: 'yes', replyAt: '2026-04-21T09:15:00Z', replyPreview: 'Hi there!' };

    describe('DMS-04 writeback integration — non-destructive skip over HTTP', () => {
      it('happy path: empty Reply → updateRow POST IS recorded', async () => {
        nextStatusResponse = { rows: [{ 'LinkedIn URL': PROSPECT_URL, 'Reply': '' }] };
        const out = await writeBackFlow(SHEET_URL, PROSPECT_URL, REPLY_FIELDS);
        assert.equal(out.skipped, false);
        const updateCalls = recordedPosts.filter(p => p.action === 'updateRow');
        assert.equal(updateCalls.length, 1, 'updateRow must be called exactly once on happy path');
        assert.equal(updateCalls[0].body.reply, 'yes');
      });

      it('canonical skip: Reply="yes" → NO updateRow POST', async () => {
        nextStatusResponse = { rows: [{ 'LinkedIn URL': PROSPECT_URL, 'Reply': 'yes' }] };
        const out = await writeBackFlow(SHEET_URL, PROSPECT_URL, REPLY_FIELDS);
        assert.equal(out.skipped, true);
        const updateCalls = recordedPosts.filter(p => p.action === 'updateRow');
        assert.equal(updateCalls.length, 0, 'updateRow MUST NOT be called when Reply="yes"');
      });

      it('value-case variants: Reply="YES" and Reply="Yes" → NO updateRow POST', async () => {
        for (const variant of ['YES', 'Yes', '  yes  ']) {
          recordedPosts = [];
          nextStatusResponse = { rows: [{ 'LinkedIn URL': PROSPECT_URL, 'Reply': variant }] };
          const out = await writeBackFlow(SHEET_URL, PROSPECT_URL, REPLY_FIELDS);
          assert.equal(out.skipped, true, `variant "${variant}" must skip`);
          assert.equal(recordedPosts.filter(p => p.action === 'updateRow').length, 0,
                       `variant "${variant}": updateRow MUST NOT be called`);
        }
      });

      it('key-case variants (Plan 11-02 revision): row.reply / row.REPLY → NO updateRow POST', async () => {
        // Plan 11-02 Task 4 normalization: statusRow['Reply'] ?? statusRow.reply
        //   ?? Object.keys(row).find(k => k.toLowerCase() === 'reply')
        const keyCases = [
          { key: 'reply',   comment: 'lowercase key (what LinkedIn/Apps Script may return)' },
          { key: 'REPLY',   comment: 'uppercase key (defensive)' },
          { key: 'ReplY',   comment: 'mixed-case key (even more defensive)' },
        ];
        for (const { key, comment } of keyCases) {
          recordedPosts = [];
          nextStatusResponse = { rows: [{ 'LinkedIn URL': PROSPECT_URL, [key]: 'yes' }] };
          const out = await writeBackFlow(SHEET_URL, PROSPECT_URL, REPLY_FIELDS);
          assert.equal(out.skipped, true, `${comment}: must skip`);
          assert.equal(recordedPosts.filter(p => p.action === 'updateRow').length, 0,
                       `${comment}: updateRow MUST NOT be called`);
        }
      });

      it('edge case: getStatus returns null / no rows → treat as empty-Reply (happy path)', async () => {
        nextStatusResponse = { rows: [] };
        const out = await writeBackFlow(SHEET_URL, PROSPECT_URL, REPLY_FIELDS);
        // No existing row → no existing Reply → write proceeds
        assert.equal(out.skipped, false);
        assert.equal(recordedPosts.filter(p => p.action === 'updateRow').length, 1);
      });
    });
    ```

    - The test uses `http.createServer` — stdlib, zero deps.
    - `getWebAppUrl()` in `src/sheets-writer.js` reads `process.env.SHEETS_WEBAPP_URL`. If it reads a different env var (e.g. `SHEETS_WEBAPP_URL` vs `GOOGLE_APPS_SCRIPT_URL`), grep `src/sheets-writer.js` for the exact name and adjust the `before()` hook accordingly.
    - The `writeBackFlow` helper in the test mirrors the orchestrator's write-back branch from Plan 11-02 Task 4 (the exact `??` lookup chain). If future refactoring extracts that branch into an exported helper on `src/linkedin/check-dms.js` (e.g. `performWriteBack`), delete the `writeBackFlow` helper and import the real one.
